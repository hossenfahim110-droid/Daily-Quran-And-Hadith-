// ====================================================== // HADITH LIBRARY ENGINE // ====================================================== const HADITH_API = "https://cdn.jsdelivr.net/gh/fawazahmed0/hadith-api@1"; // হাদিস কিতাবের নাম const hadithBookNames = { "ben-bukhari": "সহীহ বুখারী", "ben-muslim": "সহীহ মুসলিম", "ben-abudawud": "সুনান আবু দাউদ", "ben-tirmidhi": "জামে তিরমিজি", "ben-nasai": "সুনান নাসাঈ", "ben-ibnmajah": "সুনান ইবনে মাজাহ" }; // ------------------------------------------------------ // HTML নিরাপদ রাখার জন্য // ------------------------------------------------------ function escapeHTML(text) { if (!text) return ""; return String(text) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } // ------------------------------------------------------ // API Fetch Helper // ------------------------------------------------------ async function hadithFetch(url) { try { const response = await fetch(url); if (!response.ok) { throw new Error("API Error"); } return await response.json(); } catch (error) { console.error(error); throw error; } } // ------------------------------------------------------ // অধ্যায় Load // ------------------------------------------------------ async function loadHadithSections() { const book = document.getElementById("hadith-book").value; const sectionSelect = document.getElementById("hadith-section"); const result = document.getElementById("hadith-library-result"); sectionSelect.innerHTML = ``; result.innerHTML = `

অধ্যায় লোড হচ্ছে...

`; try { const url = `${HADITH_API}/editions/${book}/sections.json`; const data = await hadithFetch(url); let html = ``; for (const key in data) { if (key === "0") continue; if (data[key]) { html += ` `; } } sectionSelect.innerHTML = html; result.innerHTML = `

একটি অধ্যায় নির্বাচন করুন।

`; } catch (error) { sectionSelect.innerHTML = ``; result.innerHTML = `

অধ্যায় লোড করতে সমস্যা হয়েছে। আবার চেষ্টা করুন।

`; } } // ------------------------------------------------------ // Section / Chapter-এর সব হাদিস // ------------------------------------------------------ async function loadHadithSection(sectionId) { if (!sectionId) return; const book = document.getElementById("hadith-book").value; const result = document.getElementById("hadith-library-result"); result.innerHTML = `

হাদিসগুলো লোড হচ্ছে...

`; try { const url = `${HADITH_API}/editions/${book}/sections/${sectionId}.json`; const data = await hadithFetch(url); const hadiths = data.hadiths || []; if (!hadiths.length) { result.innerHTML = `

এই অধ্যায়ে কোনো হাদিস পাওয়া যায়নি।

`; return; } let html = ""; hadiths.forEach(hadith => { html += createHadithCard( hadith, book ); }); result.innerHTML = html; } catch (error) { console.error(error); result.innerHTML = `

হাদিস লোড করতে সমস্যা হয়েছে। আবার চেষ্টা করুন।

`; } } // ------------------------------------------------------ // নির্দিষ্ট হাদিস নম্বর // ------------------------------------------------------ async function searchHadithByNumber() { const book = document.getElementById("hadith-book").value; const number = document.getElementById("hadith-number").value.trim(); const result = document.getElementById("hadith-library-result"); if (!number) { result.innerHTML = `

হাদিস নম্বর লিখুন।

`; return; } result.innerHTML = `

হাদিস খোঁজা হচ্ছে...

`; try { const url = `${HADITH_API}/editions/${book}/${number}.json`; const data = await hadithFetch(url); /* * API response অনুযায়ী হাদিস object * বিভিন্ন version-এ structure সামান্য পরিবর্তিত * হতে পারে। তাই data.hadith থাকলে সেটি, * না থাকলে data ব্যবহার করা হচ্ছে। */ const hadith = data.hadith || data; result.innerHTML = createHadithCard( hadith, book ); } catch (error) { console.error(error); result.innerHTML = `

হাদিস নং ${escapeHTML(number)} পাওয়া যায়নি।

`; } } // ------------------------------------------------------ // Hadith Card // ------------------------------------------------------ function createHadithCard(hadith, book) { const bookName = hadithBookNames[book] || book; const number = hadith.hadithnumber || hadith.number || ""; const text = hadith.text || hadith.hadith || "হাদিসের লেখা পাওয়া যায়নি।"; return `
${escapeHTML(bookName)} হাদিস নং ${escapeHTML(number)}

${escapeHTML(text)}

উৎস: ${escapeHTML(bookName)} • হাদিস নং ${escapeHTML(number)}
`; } // ------------------------------------------------------ // প্রথমবার Bukhari-এর অধ্যায় Load // ------------------------------------------------------ loadHadithSections();
